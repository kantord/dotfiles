/*
# Desktop layout

A sidebar next to an i3 workspace.

- **sidebar** (left) — i3 workspaces on top, status cards below
  (weather, CPU/memory chart, Claude usage, volume, clock)
*/
import Wallpapers from "./components/Wallpapers.jsx";
import Sidebar from "./components/Sidebar.jsx";
import WorkspaceFrame from "./components/WorkspaceFrame.jsx";
import ThemeUnits from "./components/ThemeUnits.jsx";
import { SIDEBAR_W, TAULER_I3 } from "./lib/layout.js";

export default function render() {
  return (
    <root>
      <Wallpapers />
      {/* Panels claim space from the screen edges in declaration order; the claimed space becomes the i3 gaps. */}
      <I3Layout module={TAULER_I3}>
        <Panel id="sidebar" anchor="left" size={SIDEBAR_W}>
          <Sidebar />
        </Panel>
        <Workspaces>
          {(Contents) => <WorkspaceFrame Contents={Contents} />}
        </Workspaces>
      </I3Layout>
      <ThemeUnits />
    </root>
  );
}
